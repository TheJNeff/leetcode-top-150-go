func mergeTwoLists(list1 *ListNode, list2 *ListNode) *ListNode {
    dummy := &ListNode{}
    list3 := dummy
    for list1 != nil || list2 != nil {
        list3.Next = nil
        if list1 == nil {
            list3.Next = list2
            list2 = list2.Next
        } else if list2 == nil {
            list3.Next = list1
            list1 = list1.Next
        } else {
            if list1.Val > list2.Val {
                list3.Next = list2
                list2 = list2.Next
            } else {
                list3.Next = list1
                list1 = list1.Next
            }
        }
        list3 = list3.Next
    }
    return dummy.Next
}
